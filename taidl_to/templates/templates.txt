def require_trace(name):
    # Calls record only inside a kernel body
    if global_counters is None:
        raise ValueError(f'{name!r} must be called inside a kernel')


def generate_module(hbm_size: int, inputs: list, constants: list, outputs: list):
    if deferred:
        raise ValueError(f'loop {deferred[-1]["var"]} has no end_loop')
    _, prologue = semantic_init(hbm_size)
    params, marshal = kernel_prologue(hbm_size, inputs, constants)
    epilogue, results, result_types = kernel_epilogue(
        f'%d0.{global_counters["d0"]["counter"]}', outputs, ssa_types
    )
    returned = f' {", ".join(results)} : {", ".join(result_types)}' if results else ''
    module = 'module @compiled {\n'

    module += f'func.func public @main({", ".join(params)}) -> ({", ".join(result_types)}) '
    module += '{\n'
    for line in marshal + prologue:
        module += '\t' + line + '\n'
    for line in instruction_scopes['main'] + epilogue:
        module += '\t' + line + '\n'
    module += f'\treturn{returned}\n'
    module += '}\n}\n'

    return module


def start_loop(var_name, start, end):
    require_trace('start_loop')
    # A name a loop expression can read, not a parameter's
    if not isinstance(var_name, str):
        raise TypeError(f'start_loop {var_name!r}: loop variable {var_name!r} is not a string')
    if not (var_name.isascii() and var_name.isidentifier()):
        raise ValueError(f'start_loop {var_name!r}: loop variable {var_name!r} is not a name')
    if var_name.startswith('arg') and var_name[3:].isdigit():
        raise ValueError(
            f'start_loop {var_name!r}: %{var_name} is a name the module keeps for its parameters'
        )
    # Each bound an integer, a NumPy one as an int
    for bound, value in (('start', start), ('end', end)):
        if not integral(value):
            raise TypeError(f'start_loop {var_name!r}: {bound} {value!r} is not an integer')
    start, end = int(start), int(end)
    # A variable no enclosing loop holds
    if any(loop['var'] == '%' + var_name for loop in deferred):
        raise ValueError(f'start_loop {var_name!r}: loop variable %{var_name} is already in use')
    deferred.append({'var': '%' + var_name, 'start': start, 'end': end, 'calls': []})


def end_loop():
    require_trace('end_loop')
    if not deferred:
        raise ValueError('end_loop has no start_loop')
    loop = deferred.pop()
    # An inner loop deferred as a call of its outer one
    if deferred:
        deferred[-1]['calls'].append(loop)
    else:
        trace_loop(loop)


def defer(name, args):
    # A call of an open loop, traced once the outermost ends
    deferred[-1]['calls'].append((name, args))


def trace_loop(loop):
    # A loop with no iterations traces nothing
    if loop['start'] >= loop['end']:
        return
    # A deferred loop's calls, traced in order as one stablehlo.while
    open_loop(loop['var'], loop['start'], loop['end'])
    for call in loop['calls']:
        if isinstance(call, dict):
            trace_loop(call)
        else:
            name, args = call
            globals()[name](**args)
    close_loop()


def open_loop(var, start, end):
    global cur_scope
    loop = {
        'var': var,
        'id': global_counters['total_loop_counter'],
        'start': start,
        'end': end,
        'parent': cur_scope,
        'inits': [],
        'args': [],
        'types': [],
    }
    global_counters['total_loop_counter'] += 1
    # The body names each buffer afresh as a block argument
    for name, counter in global_counters.items():
        if isinstance(counter, dict):
            loop['inits'].append(f'%{name}.{counter["counter"]}')
            counter['counter'] += 1
            loop['args'].append(f'%{name}.{counter["counter"]}')
            loop['types'].append(mlir_type(counter['type']))
            ssa_types[loop['args'][-1]] = loop['types'][-1]
    loop_stack.append(loop)
    cur_scope = f'loop{loop["id"]}'


def close_loop():
    global cur_scope
    loop = loop_stack.pop()
    body = instruction_scopes.pop(cur_scope, [])
    cur_scope = loop['parent']
    n = f'%loop{loop["id"]}'
    yields, results = [], []
    for name, counter in global_counters.items():
        if isinstance(counter, dict):
            yields.append(f'%{name}.{counter["counter"]}')
            counter['counter'] += 1
            results.append(f'%{name}.{counter["counter"]}')
            ssa_types[results[-1]] = mlir_type(counter['type'])
    types = ', '.join(['tensor<i64>'] + loop['types'])
    pairs = ', '.join(
        f'{arg} = {init}'
        for arg, init in zip([loop['var']] + loop['args'], [f'{n}.start'] + loop['inits'])
    )
    lines = [
        f'{n}.start = stablehlo.constant dense<{loop["start"]}> : tensor<i64>',
        f'{", ".join([n + ".var"] + results)} = stablehlo.while({pairs}) : {types}',
        'cond {',
        f'\t{n}.end = stablehlo.constant dense<{loop["end"]}> : tensor<i64>',
        f'\t{n}.more = stablehlo.compare LT, {loop["var"]}, {n}.end : (tensor<i64>, tensor<i64>) -> tensor<i1>',
        f'\tstablehlo.return {n}.more : tensor<i1>',
        '} do {',
    ]
    lines += ['\t' + line for item in body for line in item.split('\n') if line.strip()]
    lines += [
        f'\t{n}.one = stablehlo.constant dense<1> : tensor<i64>',
        f'\t{n}.next = stablehlo.add {loop["var"]}, {n}.one : tensor<i64>',
        f'\tstablehlo.return {", ".join([n + ".next"] + yields)} : {types}',
        '}',
    ]
    instruction_scopes[cur_scope].append('\n'.join(lines))


def debug(prefix: str, data: str) -> None:
    require_trace('debug')
    if deferred:
        defer('debug', locals())
        return
    output = []

    # The handler prints the prefix up to a nul
    prefix_bytes = jnp.frombuffer(prefix.encode('utf-8') + b'\0', dtype=jnp.uint8)
    prefix_len = len(prefix_bytes)
    prefix_string = '['
    for i in range(len(prefix_bytes)):
        if i != 0:
            prefix_string += ', '
        prefix_string += f'{prefix_bytes[i]}'
    prefix_string += ']'

    # A buffer and an integer index of its leading dims
    data_name, debug_dims = debug_index(data, global_counters)

    rhs_loc = parameter_util(global_counters, data_name)
    data_dims = ast.literal_eval(
        '[' + global_counters[data_name]['type'].split('[')[1].split(']')[0] + ']'
    )
    starts = []
    for i in range(len(debug_dims)):
        data_dims[i] = 1
    for i in range(len(data_dims)):
        data_dims[i] = str(data_dims[i])
        if i < len(debug_dims):
            starts.append(debug_dims[i])
        else:
            starts.append(0)

    data_type = global_counters[data_name]['type'].split('[')[0]
    lhs_loc, lhs_type = lhs_util(global_counters, 'debug_slice', data_type, data_dims)
    check_window('debug', data_name, starts, mlir_type(lhs_type), ssa_types[rhs_loc])
    line = sh_slice_load(lhs_loc, lhs_type, rhs_loc, starts, ssa_types)
    output.append(line)

    rhs_loc = parameter_util(global_counters, 'debug_slice')
    lhs_loc, lhs_type = lhs_util(global_counters, 'debug_slice_f64', 'f64', data_dims)
    line = sh_cast('convert', lhs_loc, lhs_type, rhs_loc, ssa_types)
    output.append(line)

    rhs_loc = parameter_util(global_counters, 'debug_slice_f64')
    debug_prefix_loc = f'%debug_prefix.{global_counters["instruction_counter"]}'
    output.append(
        f'{debug_prefix_loc} = stablehlo.constant dense<{prefix_string}> : tensor<{prefix_len}xui8>'
    )
    output.append(
        f'%call.{global_counters["instruction_counter"]} = stablehlo.custom_call @{__package__}.print_handler({debug_prefix_loc}, {rhs_loc}) {{api_version = 4 : i32, has_side_effect = true}} : (tensor<{prefix_len}xui8>, {ssa_types[rhs_loc]}) -> tensor<i32>'
    )
    instruction_scopes[cur_scope] += output
    global_counters['instruction_counter'] += 1
